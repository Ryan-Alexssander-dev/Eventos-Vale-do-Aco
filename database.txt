CRATE DATABASE IF NOT EXIST eventos_VDA;

USER eventos_VDA;

CRATE TABLE IF NOT EXIST users(
id_user INT  UNSIGNED AUTO_INCREMENT PRIMARY KEY,
nome VARCHAR(200) NOT NULL,
e-mail VARCHAR(200) NOT NULL,
senha VARCHAR (255) NOT NULL,
telefone VARCHAR (20) NOT NULL
)ENGINE=InnoDB;

CRATE TABLE IF NOT EXIST companys(
id_company INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
cnpj VARCHAR(18) NOT NULL UNIQUE,
nome_empresa VARCHAR(255) NOT NUL,
e-mail VARCHAR(200) NOT NULL,
telefone VARCHAR (20) NOT NULL,
FOREIGN KEY(id_user) REFERENCES users(id_user)
)ENGINE=InnoDB;
